# Right Team Digital — Apache config for Hostinger (copied to .htaccess by tools/build_deploy.py)
# {{ENV}} build for {{HOST}}

Options -Indexes
DirectoryIndex index.html
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
  RewriteEngine On

  # always HTTPS
  RewriteCond %{HTTPS} off
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

  # one address: www.{{HOST}} -> {{HOST}}
  RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
  RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

  # never serve hidden files (.git, .env ...)
  RewriteRule (^|/)\. - [F]

  # ---- old site addresses -> new pages (fill in before going to production) ----
  # Google and old links still point at the previous site's pages. Send each one somewhere
  # useful instead of a 404, e.g.:
  # RewriteRule ^services/gst-registration/?$ /services.html [L,R=301]
  # RewriteRule ^contact-us/?$ /start.html [L,R=301]
{{REDIRECTS}}
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
{{ENV_HEADERS}}
  # pages: always check for a fresh copy; scripts/styles carry ?v= so they can be cached
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  <FilesMatch "\.(jpg|jpeg|png|webp|svg|ico|woff2?)$">
    Header set Cache-Control "public, max-age=2592000"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript text/javascript application/json image/svg+xml text/xml
</IfModule>
